1import { NextResponse } from 'next/server';
2import { auth } from '@/lib/auth';
3import { prisma } from '@/lib/prisma';
4import { stripe } from '@/lib/stripe';
5
6// End-to-end SaaS Multi-Tenant Provisioning
7export async function POST(req: Request) {
8  const session = await auth();
9  if (!session?.tenantId) return new NextResponse('Unauthorized', { status: 401 });
10
11  const { planId } = await req.json();
12  const checkout = await stripe.checkout.sessions.create({
13    customer: session.user.stripeCustomerId,
14    mode: 'subscription',
15    line_items: [{ price: planId, quantity: 1 }],
16    metadata: { tenantId: session.tenantId }
17  });
18  return NextResponse.json({ url: checkout.url });
19}
